function displayCart() { let cartSection = document.getElementById("cartSection"); let cartTotal = document.getElementById("cartTotal"); let orderButton = document.getElementById("placeOrderButton"); // If cart is empty if (customerCart.length === 0) { cartSection.innerHTML = `

Your Cart is Empty.

`; cartTotal.innerHTML = ""; orderButton.style.display = "none"; return; } // Maximum 10 vegetables if (customerCart.length > 10) { customerCart = customerCart.slice(0, 10); saveCart(); } cartSection.innerHTML = ""; let grandTotal = 0; // Display all vegetables customerCart.forEach(function(item, index) { let subtotal = Number(item.quantityKg) * Number(item.pricePerKg); grandTotal += subtotal; let quantityText; // Convert KG to grams for display if (Number(item.quantityKg) < 1) { quantityText = (Number(item.quantityKg) * 1000) .toFixed(0) + " g"; } else { quantityText = Number(item.quantityKg) .toFixed(3) + " KG"; } cartSection.innerHTML += `
${index + 1}. 🥦 ${item.name}

Quantity: ${quantityText}

Price: ₹${Number(item.pricePerKg).toFixed(2)} / kg

Subtotal: ₹${subtotal.toFixed(2)}

`; }); // Grand Total cartTotal.innerHTML = `
🧾 GRAND TOTAL: ₹${grandTotal.toFixed(2)}
${customerCart.length}/10 vegetables in cart
`; // Show one Place Order button orderButton.style.display = "block"; }
